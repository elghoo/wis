Panduan & Code Google Apps Script (GAS)

Salin kode Apps Script di bawah ini ke Spreadsheet Anda (menu Ekstensi > Apps Script) untuk menghubungkan formulir Spreadsheet secara live dengan Dashboard ini.

  1. Buka Google Sheets respon formulir wisuda Anda.
  2. Klik Ekstensi > Apps Script.
  3. Hapus semua kode di editor `Code.gs` dan tempelkan skrip berikut.
  4. Klik Terapkan (Deploy) > Terapkan sebagai Web App.
  5. Atur Who has access / Siapa yang memiliki akses menjadi: Anyone / Siapa Saja.
  6. Salin Web App URL dan tempelkan di kotak input Web App URL pada bagian atas dashboard.
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  
  var result = [];
  if (data.length > 1) {
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if(!row[0] && !row[3]) continue; // Skip empty rows
      
      var item = {
        timestamp: row[0] ? String(row[0]) : '',
        wa: row[1] ? String(row[1]) : '',
        gender: row[2] ? String(row[2]) : '',
        fullName: row[3] ? String(row[3]) : '',
        city: row[4] ? String(row[4]) : '',
        province: row[5] ? String(row[5]) : '',
        wisudawanAttendance: row[6] ? String(row[6]) : '',
        waliAttendance: row[7] ? String(row[7]) : '',
        wali1Name: row[8] ? String(row[8]) : '',
        wali1Relation: row[9] ? String(row[9]) : '',
        wali2Name: row[10] ? String(row[10]) : '',
        wali2Relation: row[11] ? String(row[11]) : '',
        email: row[12] ? String(row[12]) : '',
        docId: row[13] ? String(row[13]) : ('DOC-' + (1000 + i))
      };
      result.push(item);
    }
  }
  
  var jsonOutput = JSON.stringify(result);
  
  // JSONP support to bypass CORS in iframe/blob origins
  if (e && e.parameter && e.parameter.callback) {
    return ContentService.createTextOutput(e.parameter.callback + '(' + jsonOutput + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  
  return ContentService.createTextOutput(jsonOutput)
    .setMimeType(ContentService.MimeType.JSON);
}
